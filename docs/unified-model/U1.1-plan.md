Статус: план, код не менялся. Задача из `docs/markdown-unified-model-plan.md`, этап 1. Только типы: без cmark, AppKit, WebKit, SwiftUI, без логики разбора.

Когда `EditorDocument` получит модель (U4.1), в `Scripts/verify-release.sh` и `Scripts/run-acceptance.sh` придётся добавить в списки `Models/MarkdownDocumentModel.swift`, `Models/MarkdownSyntaxOptions.swift`, `Models/MarkdownBlockIndex.swift`, а также разрешить зависимость от `MarkdownCalloutKind`, `CriticMarkupKind`, `MarkdownNoteEmbedItem.Kind`. ### 2.3. Что в модели НЕ хранится (выводится адаптерами из дерева)

Плоские массивы `MarkdownPresentation`, которые целиком получаются обходом дерева: `images`, `links`, `formulas`, `inlineFormulas`, `diagrams`, `audioBlocks`, `iframeBlocks`, `detailsBlocks`, `htmlBlocks`, `pageBreaks`, `ruby`, `wikiLinks`, `tables`, `codeBlocks`, `tableOfContents`, `completedTaskRanges`, `taskCheckboxes`, `listMarkers`, `paragraphs`, `headings`, `spans`. Хранить их второй раз в модели значило бы держать два источника правды. Исключение — то, что дерево выразить не может (см. корень: `unreliableRanges`, индексы, `fence`, `emoji`, `looseIndents`, `smartReplacements`).

## 3. Объявления типов

Готовый код для переноса в `LacEditor/Models/MarkdownDocumentModel.swift`. Проверен на Linux: `swiftc -frontend -parse` и `swiftc -typecheck -swift-version 6` вместе с заглушками объявлений из 2.1 дали 0 ошибок и 0 предупреждений (в том числе автоматическая проверка `Equatable` и `Sendable`).

```swift
import Foundation

/// Результат одного разбора Markdown: описывает исходный текст, позиции — UTF-16 исходника.
/// Из модели два адаптера строят HTML предпросмотра и `MarkdownPresentation` редактора.
struct MarkdownDocumentModel: Equatable, Sendable {
    /// Что считать при разборе. Соответствует флагам `include*` у `MarkdownPresentationParser.parse`
    /// и параметрам `wiki`, `critic`, `embeds` у `MarkdownRenderer.renderPage`.
    struct Includes: OptionSet, Equatable, Sendable {
        let rawValue: Int
        init(rawValue: Int) { self.rawValue = rawValue }
        /// `[[…]]` — узлы wiki-ссылок (`includeWikiLinks`; в предпросмотре — когда есть `wiki`).
        static let wikiLinks = Includes(rawValue: 1 << 0)
        /// `![[…]]` — узлы встраиваний (в предпросмотре — когда есть `embeds` или `wiki`).
        static let noteEmbeds = Includes(rawValue: 1 << 1)
        /// `{++ ++}`, `{-- --}`, `{~~ ~> ~~}`, `{>> <<}`, `{== ==}` (в предпросмотре — когда есть `critic`).
        static let criticMarkup = Includes(rawValue: 1 << 2)
        /// `emoji`: кандидаты `:имя:`, места автодополнения, участки умных замен (`includeEmojiCandidates`).
        static let emoji = Includes(rawValue: 1 << 3)
        /// `uppercaseRuns` у заголовков (`includeUppercaseRuns`).
        static let uppercaseRuns = Includes(rawValue: 1 << 4)
        /// `readableText` (`includeReadableText`).
        static let readableText = Includes(rawValue: 1 << 5)
        /// `elements`: индекс элементов для строки состояния.
        static let elementIndex = Includes(rawValue: 1 << 6)
        /// `blockIndex`: верхнеуровневые блоки для локального разбора (сегодня — колбэк `blocks:`).
        static let blockIndex = Includes(rawValue: 1 << 7)
        /// Данные только для TextKit: `smartReplacements`, `looseIndents`, `fence`.
        static let editorContext = Includes(rawValue: 1 << 8)
    }

    struct Options: Equatable, Sendable {
        /// `strictMode`, `inlineMath`, `legacyInlineMath`, `smartPunctuation` и правила пробелов задаёт `syntax`.
        var syntax: MarkdownSyntaxOptions
        var includes: Includes
        /// Заметка хранилища: `%%…%%` — комментарии (`hidesObsidianComments || appliesTo(documentURL:)`).
        var isVaultNote: Bool
        /// `detectsFrontMatter` у `parse`; у `renderPage` всегда `true`.
        var detectsFrontMatter: Bool
    }

    /// Участок текста, который перед cmark заменяется служебными символами той же длины и с теми же переводами строк.
    struct Mask: Equatable, Sendable {
        enum Kind: Equatable, Sendable {
            case obsidianComment
            case critic(CriticMarkupKind)
            case wikiLink
            case noteEmbed
        }
        var kind: Kind
        var range: NSRange
        /// Разделители: `%%`/`%%`; `{++`/`++}`, у подмены ещё `~>`; `[[`/`]]`; `![[`/`]]`.
        var markerRanges: [NSRange]
    }

    struct FrontMatter: Equatable, Sendable {
        /// От начала текста до конца закрывающей строки (без перевода строки).
        var range: NSRange
        /// Открывающий и закрывающий разделители (`---`, `---`/`...`).
        var markers: [NSRange]
    }

    /// Определение ссылки `[метка]: адрес "заголовок"`; cmark такие абзацы потребляет и наружу не отдаёт.
    struct LinkDefinition: Equatable, Sendable {
        var range: NSRange
        var label: String
    }

    /// Контекст для команд «Вставить блок кода»: куда нельзя ставить ограждение (поля `fence*` у `MarkdownPresentation`).
    struct FenceContext: Equatable, Sendable {
        var containerLines: [MarkdownFenceContainerLine]
        var blockedRanges: [NSRange]
        var blockedEOF: Bool
        var isAvailable: Bool
    }

    struct EmojiData: Equatable, Sendable {
        var candidates: [MarkdownEmojiCandidate]
        var completionRanges: [NSRange]
        var smartTextRanges: [NSRange]
        /// `nil` — границы сырого HTML подтвердить не удалось.
        var smartRawHTMLRanges: [NSRange]?
    }

    /// Номер версии текста (`EditorDocument.textRevision`); `nil` — текст не принадлежит документу (экспорт, тесты).
    var revision: UInt?
    /// Длина исходного текста в UTF-16.
    var textLength: Int
    var options: Options
    var blocks: [MarkdownBlockNode]
    var frontMatter: FrontMatter?
    var masks: [Mask]
    /// Строки, где позиции cmark не подтверждены: отсортированы и слиты.
    var unreliableRanges: [NSRange]
    /// `id` заголовков в порядке документа (без заголовков в таблицах, картинках и определениях сносок).
    var headingIDs: [String]
    var namedAnchors: [MarkdownNamedAnchor]
    var footnotes: [MarkdownFootnote]
    var linkDefinitions: [LinkDefinition]
    var looseIndents: [MarkdownLooseIndent]
    var smartReplacements: [MarkdownTextReplacement]
    var fence: FenceContext?
    var emoji: EmojiData?
    var readableText: MarkdownReadableText?
    var blockIndex: MarkdownBlockIndex?
    var elements: MarkdownElementIndex?
}

// MARK: - Блоки

struct MarkdownBlockNode: Equatable, Sendable {
    enum Kind: Equatable, Sendable {
        case paragraph
        case heading(Heading)
        case quote
        case callout(Callout)
        case alert(Alert)
        case list(ListBlock)
        case listItem(ListItem)
        case codeBlock(CodeBlock)
        case mathBlock(source: String)
        case table(Table)
        case html(HTMLBlock)
        case tableOfContents
        case footnoteDefinition(label: String)
        case frontMatter
        case thematicBreak
        case noteEmbed(NoteEmbed)
        case comment
    }

    struct Heading: Equatable, Sendable {
        enum Style: Equatable, Sendable { case atx, setext }
        var level: Int
        var style: Style
        /// Видимый текст заголовка, как `MarkdownPresentationParser.headingText`.
        var title: String
        /// Якорь `lac-heading-…`; пустой у заголовка в таблице, картинке или определении сноски.
        var id: String
        var slug: String
        /// Текст без маркеров (`MarkdownHeading.plainTextRange`).
        var contentRange: NSRange
        var emoji: [MarkdownHeading.Emoji]
        /// `nil` — не запрошено (`Includes.uppercaseRuns`) или построить не удалось.
        var uppercaseRuns: [MarkdownHeading.UppercaseRun]?
        /// Лежит прямо в документе: не в цитате, списке и не в определении сноски.
        var isTopLevel: Bool
        var inUnsupportedContainer: Bool
    }

    /// Выноска Obsidian `> [!тип]± заголовок`. Заголовок — `inlines` узла, содержимое — `children`.
    struct Callout: Equatable, Sendable {
        enum Fold: Equatable, Sendable { case none, expanded, collapsed }
        /// Тип как написан (`warning`, `tldr`).
        var type: String
        /// `nil` — неизвестный тип; оформляется как `.note` с подписью `type`.
        var kind: MarkdownCalloutKind?
        var fold: Fold
        /// `[!тип]±`.
        var markerRange: NSRange
        /// Заголовок после маркера; `nil` — пустой (подпись берётся из типа).
        var titleRange: NSRange?
    }

    /// Алерт GitHub `> [!NOTE]` (только в документе, без вложения).
    enum Alert: String, Equatable, Sendable {
        case note = "NOTE", tip = "TIP", important = "IMPORTANT", warning = "WARNING", caution = "CAUTION"
    }

    struct ListBlock: Equatable, Sendable {
        enum Marker: Equatable, Sendable {
            case bullet(UInt16)
            case ordered(start: Int, delimiter: UInt16)
        }
        var marker: Marker
        /// Плотный список (без пустых строк между пунктами).
        var isTight: Bool
        /// Вложенность списков от 1.
        var depth: Int
    }

    /// Пункт списка; маркер — `markerRanges[0]`.
    struct ListItem: Equatable, Sendable {
        /// Номер у нумерованного списка.
        var ordinal: Int?
        var task: MarkdownTaskCheckbox?
    }

    struct CodeBlock: Equatable, Sendable {
        var isFenced: Bool
        /// Строка info без обрамления, как `cmark_node_get_fence_info`.
        var info: String
        var infoRange: NSRange?
        /// Есть закрывающее ограждение.
        var isClosed: Bool
        /// Тело по строкам: без префиксов контейнеров и без ограждений, без перевода строки.
        var bodySegments: [NSRange]
        /// Сколько пробелов у каждого отрезка добавлено при раскрытии табов (как `MarkdownCodeBlock.bodyPadding`).
        var bodyPadding: [Int]
        /// Тело как `cmark_node_get_literal`.
        var literal: String
        /// `mermaid`, `flow`, `sequence` у блока верхнего уровня с подтверждённой позицией.
        var diagram: MarkdownDiagram.Kind?
    }

    struct Table: Equatable, Sendable {
        struct Cell: Equatable, Sendable {
            /// Между разделителями `|`, с пробелами по краям.
            var range: NSRange
            /// Без обрамляющих пробелов.
            var contentRange: NSRange
            var column: Int
            /// Сколько `|` ячейка поглотила при слиянии (`CMARK_OPT_TABLE_SPANS`).
            var virtualSeparators: Int
            var inlines: [MarkdownInlineNode]
        }
        struct Row: Equatable, Sendable {
            var range: NSRange
            var isHeader: Bool
            var cells: [Cell]
        }
        var alignments: [MarkdownTableAlignment]
        /// Строка-разделитель `| --- | :-: |`.
        var delimiterRange: NSRange
        var rows: [Row]
    }

    struct HTMLBlock: Equatable, Sendable {
        struct Details: Equatable, Sendable {
            var summary: String
            var body: String
            var summaryHTML: String?
            var bodyHTML: String?
            var isOpen: Bool
        }
        struct Frame: Equatable, Sendable {
            var source: URL
            var width: String?
            var height: String?
            var title: String?
            var fallback: String
            var allowsFullscreen: Bool
            var scrolling: String?
            var frameBorder: String?
            var allowTransparency: String?
            /// Подтверждённый исходный HTML и его вид в редакторе.
            var html: String
            var displayHTML: String
        }
        struct Twitter: Equatable, Sendable {
            var id: String
            var postURL: String
            var text: String
        }
        enum Kind: Equatable, Sendable {
            /// Не подтверждён как особый вид: выводится как сырой HTML (с `tagfilter`).
            case raw
            /// Подтверждённый блок: `displayHTML` — то, что редактор показывает как HTML-блок.
            case confirmed(displayHTML: String)
            case details(Details)
            case media(MarkdownEmbeddedMedia)
            case iframe(Frame)
            case svg(Data)
            case pageBreak(style: String)
            case twitter(Twitter)
            case images([MarkdownInlineNode.Image])
        }
        /// Текст блока, как `cmark_node_get_literal`.
        var literal: String
        var kind: Kind
    }

    struct NoteEmbed: Equatable, Sendable {
        var target: String
        var heading: String?
        var alias: String?
        var kind: MarkdownNoteEmbedItem.Kind
    }

    /// Строка блока внутри контейнеров: префикс стоит в начале `line`.
    struct ContainerPrefix: Equatable, Sendable {
        var line: NSRange
        var prefixLength: Int
        /// Чем префикс заменяется на строках продолжения (`>` остаётся, `- ` становится пробелами).
        var continuation: String
    }

    var kind: Kind
    var range: NSRange
    var markerRanges: [NSRange]
    var containerPrefixes: [ContainerPrefix]
    var children: [MarkdownBlockNode]
    var inlines: [MarkdownInlineNode]
    var isUnreliable: Bool
}

// MARK: - Строчные элементы

struct MarkdownInlineNode: Equatable, Sendable {
    enum Kind: Equatable, Sendable {
        case text(String)
        case code(String)
        case strong, emphasis, strikethrough

        case inlineStyle(MarkdownInlineStyle)

        case image(Image)
        case media(MarkdownEmbeddedMedia)
        case iframe(MarkdownBlockNode.HTMLBlock.Frame)
        case svg(Data)
        case ruby(Ruby)
        case wikiLink(WikiLink)
        case noteEmbed(MarkdownBlockNode.NoteEmbed)
        case formula(source: String)
        case footnoteReference(label: String)
        case emoji(name: String)

        case tag(String)
        case softBreak, hardBreak
        case rawHTML(String)

    struct Link: Equatable, Sendable {
        enum Form: Equatable, Sendable { case inline, reference, autolink, html }
        var url: String
        var title: String?
        var form: Form
        /// Адрес как написан; `nil` у ссылочной формы и автоссылки.

    }

    struct Image: Equatable, Sendable {
        enum Syntax: Equatable, Sendable { case markdown, reference, html }
        var url: String
        var alt: String
        var title: String?
        var syntax: Syntax
        var addressRange: NSRange?
        /// `width`, `height`, `style` у `<img>`.
        var sizing: MarkdownImageSizing?
        /// Единственный элемент абзаца верхнего уровня.
        var isBlock: Bool
    }

    struct Ruby: Equatable, Sendable {

        var reading: String
        var baseRange: NSRange
        var readingRange: NSRange

        var heading: String?
        var alias: String?
        var displayRange: NSRange

    var range: NSRange
    var markerRanges: [NSRange]

}
```

### 3.1. Правила, которые код выразить не может

Эти соглашения записываются комментариями `///` у типов и действуют для U2.3 (проверяются тестом U2.6).

1. **Позиции.** `NSRange` в UTF-16 исходника (как везде в проекте; `NSRange` — `Sendable`, поля `NSRange` в `Sendable`-структурах уже есть в `MarkdownPresentation`). `range` пустым не бывает, кроме `.mathBlock` и `.codeBlock` без тела.
2. **Диапазон узла.** Для блока — от первого значащего символа первой строки (после префиксов контейнера) до конца последней строки без перевода строки; для кода, формулы, таблицы и HTML-блока — целые строки, как сегодня дают `fenceRange`, `mathBlockRange`, `tableRange`. Строчные — как у соответствующих spans сегодня. Равенство с нынешними диапазонами проверяют эталоны U0.2.
3. **Узел без подтверждённой позиции.** `range` равен диапазону контейнера (как `confirmed ?? container` в `MarkdownReadableTextParser`), `isUnreliable == true`. Опционального `range` нет: адаптеры не должны обрабатывать `nil` на каждом шаге.
4. **`markerRanges` — служебные знаки, которые редактор скрывает вне курсора**, в порядке текста: заголовок ATX — `#…`, setext — линия подчёркивания; цитата — по одному `>` на строку; пункт — маркер списка; блок кода — открывающее и закрывающее ограждение (закрывающего нет, если `isClosed == false`); `---` — вся линия; front matter — `---`/`---`; ссылка `[` и `](адрес…)`; изображение `![` и `](…)`; strong/emphasis/strikethrough/code/highlight/sub/sup — открывающий и закрывающий знак; underline/kbd/inline style/ссылка из HTML — открывающий и закрывающий тег; critic — открывающий, у подмены ещё `~>`, закрывающий; комментарий — `%%` и `%%`; wiki — как `MarkdownWikiLink.markerRanges`; встраивание — `![[` и `]]`; выноска — `[!тип]±` и `>` по строкам; алерт — `[!NOTE]` и `>` по строкам; жёсткий разрыв — `\` или два пробела или `<br>`.
5. **`children` и `inlines`.** `children` — вложенные блоки (цитата, выноска, алерт, список, пункт, определение сноски). `inlines` — строчное содержимое листового блока (абзац, заголовок, заголовок выноски). У остальных блоков оба массива пусты. Ячейка таблицы хранит свои `inlines` сама. Узел `.critic` подмены держит все дочерние рядом: сторону «было»/«стало» определяет положение относительно `markerRanges[1]`.
6. **`containerPrefixes`.** Заполняются только у листовых блоков внутри цитат и пунктов, по строке на каждую строку блока (в том числе первую); у блоков верхнего уровня пусто. Контейнеры (`quote`, `list`, `listItem`) префиксов не хранят: их маркеры лежат в `markerRanges`. `FenceContext.containerLines` — другое: узкий набор строк для вставки ограждения, перенесён из разбора как есть.
7. **Изображения** листьев не имеют: подпись — `alt`, дочерних узлов нет. **Блочная формула** `$$…$$` — блок `.mathBlock`, строчная — `.formula`.
8. **Диаграммы** не отдельный вид блока, а поле `CodeBlock.diagram` (см. спорные решения). Тело диаграммы и обычного кода получается одинаково.
9. **Заголовки** вне таблиц, картинок и определений сносок получают `id` из `MarkdownHeadingAnchors.Allocator` в порядке документа, `headingIDs` — тот же порядок. Якоря `lac-anchor-N`, номера заголовков и оглавление строит HTML-адаптер.
10. **Маскированные участки.** Каждый элемент `masks` соответствует узлу дерева (`.comment`, `.critic`, `.wikiLink`, `.noteEmbed`); `masks` — плоский отсортированный список для быстрого поиска, дерево — для обхода. Тест U2.6 проверяет, что они согласованы.
11. **Опциональность.** Поля, которые считаются только по `options.includes`, — `Optional` (список ниже). Если флаг не запрошен, поле `nil`; потребитель обязан проверять флаг, а не угадывать по пустоте.

### 3.3. Функции поиска

В U1.1 не добавляются. Адаптеры U3 и U5 обходят дерево подряд. Двоичный поиск по позиции нужен только `EditorCoordinator` (U6.3: Cmd-клик, таблица под курсором, сноска); сегодня он обходится готовыми массивами `MarkdownPresentation` и `MarkdownElementIndex.path(at:)`. Если U6.3 потребует поиск по дереву, функция `MarkdownDocumentModel.topLevelBlockIndex(containing:)` добавляется там вместе с тестом (блоки верхнего уровня отсортированы по `range.location`, не пересекаются).

## 5. Решения и допущения

1. Модель описывает только исходный текст; сам текст в ней не хранится (длина — `textLength`). Адаптер получает текст рядом с моделью.
2. Везде `struct` и `enum`, без классов; все поля — `var`, чтобы U6.2 (склейка) могла сдвигать диапазоны без пересоздания.
3. Тексты внутри узлов (`text`, `literal`, `title`) хранятся строками. Это удваивает память для кода и текста; цену измеряют замеры P1–P9 (U9.3). Если выйдет дорого, `.text` заменяют диапазоном с флагом «совпадает с исходником» (поле `Piece.matchesSource` уже так работает).
4. `Equatable` у корня нужен для сверки моделей в тестах (LF и CRLF, склейка и полный разбор). Сравнение дорогое, на горячем пути не вызывается.
5. Порядок объявлений в файле: корень, блоки, строчные; без `extension`, без методов (кроме синтезированных). Комментарии `///` на русском над неочевидным, как в проекте.

## 8. Спорные решения

1. **Диаграмма — поле блока кода, а не вид блока.** Мастер-план перечисляет «диаграмму» отдельным видом. Но и редактор, и рендер сначала разбирают обычный блок кода (тело, ограждения, `isClosed`), а `mermaid/flow/sequence` — только его классификация по `info`. Отдельный вид потребовал бы дублировать `CodeBlock`. Если основной агент настаивает на виде, замена механическая: вынести поле в `case diagram(CodeBlock, kind:)`.
2. **`MarkdownSyntaxOptions` переносится в свой файл (изменение чужого файла).** Иначе модель зависит от cmark через `MarkdownMathExtension.swift`. Альтернатива: оставить тип на месте до U4.1; тогда модель нельзя будет добавить в `Scripts/*` без cmark.
3. **Правка двух чужих типов** (`MarkdownReadableText`, `MarkdownLooseIndent`): добавляется `Equatable`, иначе корень нельзя сделать `Equatable`. Альтернатива без правок: собственный `LooseIndent` в модели и ручной `==` у корня; хуже: дубль типа и ручное сравнение.
4. **`MarkdownCalloutKind` и `MarkdownNoteEmbedItem.Kind` остаются в файлах, которые ссылаются на разбор.** На стенде это не мешает; для `Scripts/*` в U4.1 нужен список выше. Либо вынести два типа в `Models/` в U4.1, либо в U1.1 завести дубли (отвергнуто).
5. **`isUnreliable` вместо опционального `range`.** Узел без подтверждённой позиции получает диапазон контейнера (как читаемый текст сейчас). Так проще адаптерам, но их нужно приучить проверять `isUnreliable` перед тем, как привязывать к диапазону оформление. Альтернатива `NSRange?` слегка безопаснее, но ломает синтез `Equatable` не так, как хочется, и заставляет все обходы разворачивать опционал.
6. **`uppercaseRuns` лежат в узле заголовка, а не в корне.** Мастер-план перечисляет их среди полей корня, но они всегда принадлежат одному заголовку (так хранит и `MarkdownHeading`). Корень их не дублирует.
7. **Дублирование между `masks` и узлами `.comment/.critic/.wikiLink/.noteEmbed`.** `masks` нужны, чтобы проверить U2.2 и быстро найти исключаемые участки без обхода дерева; узлы нужны адаптерам. После U2.3 можно убрать `masks`, если ни один потребитель не пойдёт по ним (решает U5a).
8. **`containerPrefixes` только у листовых блоков.** Хранить префиксы у каждого контейнера значит повторять их по глубине. Расплата: вопрос «какой префикс у строки N блока `.listItem`» решается через потомков.
9. **Определения ссылок (`linkDefinitions`) cmark не отдаёт.** Источник — логика `recoverReferenceParagraphs` и метки таблиц. Поле объявлено сразу, но U2.3 должен убедиться, что данных хватает; если нет, поле остаётся пустым и вопрос уходит в U6.4 (таблицы с `referenceLabels`).
10. **`.tag(String)` объявлен, хотя разбор сейчас тегов не строит** (их ищет `MarkdownInlineTags` по текстовым узлам). Это вид из мастер-плана; если U7.2 решит читать теги из `.text`, вид удаляется.
11. **`HTMLBlock.Frame`, `Details`, `Twitter` копируют поля сервиса `MarkdownHTMLMediaService`.** Подтверждение HTML (SwiftSoup) остаётся в ядре U2.3, а модель хранит только результат. Любое новое поле у `EmbeddedFrame` потребует правки двух мест; зафиксируем в `docs/agents/markdown-html-media.md` при реализации U2.3.
12. **Нет функций поиска** (раздел 3.3).
